# Clean ACS place names
acs_clean <- acs_data %>%
  separate(NAME, into = c("city_raw", "state_name"), sep = ", ") %>%
  mutate(
    city = city_raw %>%
      str_remove(" city") %>%
      str_remove(" city \\(balance\\)") %>%
      str_remove(" town") %>%
      str_remove(" CDP") %>%
      str_remove(" municipality") %>%
      str_remove(" urban county") %>%
      str_remove(" metropolitan government \\(balance\\)") %>%
      str_remove(" consolidated government \\(balance\\)"),
    state = state.abb[match(state_name, state.name)]
  )

# Fix name issues from csv
acs_clean <- acs_clean %>%
  mutate(
    city = case_when(
      city == "St. Paul" ~ "Saint Paul",
      city == "San Jose" ~ "San José",
      str_detect(city_raw, "Washington") & state_name == "District of Columbia" ~ "Washington",
      str_detect(city_raw, "Urban Honolulu") ~ "Honolulu",
      str_detect(city_raw, "Nashville-Davidson") ~ "Nashville",
      str_detect(city_raw, "Louisville/Jefferson County") ~ "Louisville",
      TRUE ~ city
    ),
    state = ifelse(state_name == "District of Columbia", "DC", state)
  )
# 3 economic outcome variables
acs_economic <- acs_clean %>%
  transmute(
    city,
    state,
    median_income = median_incomeE,
    poverty_rate = (poverty_countE / poverty_totalE) * 100,
    unemployment_rate = (unemployedE / labor_forceE) * 100,
    population = populationE
  )
# Merge ACEEE data with ACS data
final_data <- policy_data %>%
  left_join(acs_economic, by = c("city", "state"))
